FROM php:8.2-fpm

RUN apt-get update && apt-get install -y \
    nginx \
    git \
    curl \
    libpng-dev \
    libonig-dev \
    libxml2-dev \
    zip \
    unzip \
    supervisor

RUN apt-get clean && rm -rf /var/lib/apt/lists/*

RUN docker-php-ext-install pdo_mysql mbstring exif pcntl bcmath gd

COPY --from=composer:latest /usr/bin/composer /usr/bin/composer

WORKDIR /var/www/html

COPY . .

RUN composer install --no-interaction --optimize-autoloader --no-dev

RUN chown -R www-data:www-data /var/www/html/storage /var/www/html/bootstrap/cache
RUN chmod -R 775 /var/www/html/storage /var/www/html/bootstrap/cache

RUN echo "server {" > /etc/nginx/sites-enabled/default
RUN echo "    listen 10000;" >> /etc/nginx/sites-enabled/default
RUN echo "    server_name _;" >> /etc/nginx/sites-enabled/default
RUN echo "    root /var/www/html/public;" >> /etc/nginx/sites-enabled/default
RUN echo "    index index.php;" >> /etc/nginx/sites-enabled/default
RUN echo "    location / {" >> /etc/nginx/sites-enabled/default
RUN echo "        try_files \$uri \$uri/ /index.php?\$query_string;" >> /etc/nginx/sites-enabled/default
RUN echo "    }" >> /etc/nginx/sites-enabled/default
RUN echo "    location ~ \.php$ {" >> /etc/nginx/sites-enabled/default
RUN echo "        fastcgi_pass 127.0.0.1:9000;" >> /etc/nginx/sites-enabled/default
RUN echo "        fastcgi_index index.php;" >> /etc/nginx/sites-enabled/default
RUN echo "        fastcgi_param SCRIPT_FILENAME \$document_root\$fastcgi_script_name;" >> /etc/nginx/sites-enabled/default
RUN echo "        include fastcgi_params;" >> /etc/nginx/sites-enabled/default
RUN echo "    }" >> /etc/nginx/sites-enabled/default
RUN echo "}" >> /etc/nginx/sites-enabled/default

RUN echo "[supervisord]" > /etc/supervisor/conf.d/supervisord.conf
RUN echo "nodaemon=true" >> /etc/supervisor/conf.d/supervisord.conf
RUN echo "" >> /etc/supervisor/conf.d/supervisord.conf
RUN echo "[program:php-fpm]" >> /etc/supervisor/conf.d/supervisord.conf
RUN echo "command=php-fpm" >> /etc/supervisor/conf.d/supervisord.conf
RUN echo "autostart=true" >> /etc/supervisor/conf.d/supervisord.conf
RUN echo "autorestart=true" >> /etc/supervisor/conf.d/supervisord.conf
RUN echo "" >> /etc/supervisor/conf.d/supervisord.conf
RUN echo "[program:nginx]" >> /etc/supervisor/conf.d/supervisord.conf
RUN echo "command=nginx -g 'daemon off;'" >> /etc/supervisor/conf.d/supervisord.conf
RUN echo "autostart=true" >> /etc/supervisor/conf.d/supervisord.conf
RUN echo "autorestart=true" >> /etc/supervisor/conf.d/supervisord.conf

EXPOSE 10000

CMD ["/usr/bin/supervisord", "-c", "/etc/supervisor/conf.d/supervisord.conf"]
